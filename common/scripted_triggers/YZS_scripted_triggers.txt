# 国家 scope：生日按当前游戏日历判断，不依赖开局日期或每日累加变量。
# global.year 是引擎当前年份；不要替换成千禧自行维护的 global.month。
# 半开区间分别覆盖 6 月 8 日全天，以及 6 月 9 日至 28 日的 20 天补庆期。
yz_sana_is_birthday = {
	custom_trigger_tooltip = {
		tooltip = sana_birthday_date_tt
		meta_trigger = {
			text = {
				NOT = { date < "[YEAR].6.8" }
				date < "[YEAR].6.9"
			}
			YEAR = global.year
		}
	}
}

yz_sana_is_birthday_makeup_window = {
	custom_trigger_tooltip = {
		tooltip = sana_birthday_makeup_date_tt
		meta_trigger = {
			text = {
				NOT = { date < "[YEAR].6.9" }
				date < "[YEAR].6.29"
			}
			YEAR = global.year
		}
	}
}

# 三个生日入口共用国家级年度记录；首次庆祝前变量不存在，视为本年尚未庆祝。
# 加成到期只清理原有 sana_birthday_succeed，不清除此年份；跨年直接与 global.year 比较。
# 调试开关在手动调用者的 OR 分支绕过日期和年度限制，不影响自动任务的日历规则。
yz_sana_birthday_not_celebrated_this_year = {
	custom_trigger_tooltip = {
		tooltip = sana_birthday_not_celebrated_this_year_tt
		OR = {
			NOT = { has_variable = yz_sana_last_birthday_year }
			NOT = { check_variable = { yz_sana_last_birthday_year = global.year } }
		}
	}
}
