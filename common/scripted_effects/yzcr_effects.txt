yzcr_open_window = {
    if = {
        limit = { OR = { has_country_flag = have_yoshino has_country_flag = have_mako has_country_flag = have_murasame has_country_flag = have_lena } }
        set_country_flag = yzcr_main_complete
    }
	set_variable = { yzcr_work = 1 }
	if = {
		limit = {
			OR = {
				NOT = { has_country_flag = yzcr_initialized }
				check_variable = { var = yzcr_step value = 100 compare = equals }
			}
		}
		yzcr_begin = yes
	}
	else = {
		clr_country_flag = yzcr_lang_menu
		clear_variable = yzcr_art_lang
		yzcr_refresh_background = yes
		set_country_flag = yzcr_open
		clr_country_flag = yzcr_help
		add_to_variable = { yzcr_dirty = 1 }
	}
}

yzcr_resolve = {
	set_variable = { yzcr_step = 100 }
	set_variable = { yzcr_result = 0 }
	if = {
		limit = {
			check_variable = { var = yzcr_q3 value = 1 compare = equals }
			check_variable = { var = yzcr_q4 value = 1 compare = equals }
			check_variable = { var = yzcr_q5 value = 2 compare = equals }
		}
		set_variable = { yzcr_result = 4 }
	}
	else_if = {
		limit = {
			check_variable = { var = yzcr_q4 value = 1 compare = equals }
			check_variable = { var = yzcr_q5 value = 1 compare = equals }
			check_variable = { var = yzcr_q8 value = 1 compare = equals }
		}
		set_variable = { yzcr_result = 1 }
	}
	else_if = {
		limit = {
			check_variable = { var = yzcr_q2 value = 2 compare = equals }
			check_variable = { var = yzcr_q4 value = 2 compare = equals }
		}
		set_variable = { yzcr_result = 2 }
	}
	else_if = {
		limit = {
			check_variable = { var = yzcr_q4 value = 3 compare = equals }
			check_variable = { var = yzcr_q6 value = 2 compare = equals }
		}
		set_variable = { yzcr_result = 3 }
	}
	else_if = {
		limit = {
			check_variable = { var = yzcr_q1 value = 1 compare = equals }
			check_variable = { var = yzcr_q7 value = 2 compare = equals }
		}
		if = {
			limit = { has_country_flag = yzcr_main_complete }
			set_variable = { yzcr_step = 9 }
		}
		else = {
			set_variable = { yzcr_result = -1 }
		}
	}
}

            yzcr_recruit_chisaki = yes
        }
        else_if = {
            limit = { check_variable = { var = yzcr_rj_result value = 0 compare = equals } }

            yzcr_recruit_hazuki = yes

# Shared saved preference: 0 ask, 1 SD, 2 routes; pending work is 1 Senren / 2 RJ.
yzcr_dispatch_request = {
    yzcr_close_window = yes
    clr_country_flag = QL_recruit_show
    clr_country_flag = RJ_recruit_show
    set_country_flag = yzcr_prompt_pending
    if = {
        limit = { check_variable = { var = yzcr_preference value = 1 compare = equals } }
        yzcr_choose_legacy = yes
    }
    else_if = {
        limit = { check_variable = { var = yzcr_preference value = 2 compare = equals } }
        yzcr_choose_routes = yes
    }
    else = {
        country_event = { id = yz_choice_recruit.1 }
    }
}
yzcr_choose_legacy = {
    if = {
        limit = { yzcr_recruitment_allowed = yes has_country_flag = yzcr_prompt_pending }
        if = {
            limit = { check_variable = { var = yzcr_pending_work value = 2 compare = equals } }
            set_country_flag = RJ_recruit_show
        }
        else = { set_country_flag = QL_recruit_show }
        clr_country_flag = yzcr_prompt_pending
        clear_variable = yzcr_pending_work
    }
}
yzcr_choose_routes = {
    if = {
        limit = { yzcr_recruitment_allowed = yes has_country_flag = yzcr_prompt_pending }
        if = {
            limit = { check_variable = { var = yzcr_pending_work value = 2 compare = equals } }
            yzcr_rj_open_window = yes
        }
        else = { yzcr_open_window = yes }
        clr_country_flag = yzcr_prompt_pending
        clear_variable = yzcr_pending_work
    }
}
yzcr_request_senren = {
    if = {
        limit = { yzcr_recruitment_allowed = yes NOT = { has_country_flag = yzcr_prompt_pending } }
        if = {
            limit = { has_country_flag = QL_recruit_show }
            clr_country_flag = QL_recruit_show
        }
        else_if = {
            limit = { has_country_flag = yzcr_open yzcr_active = yes }
            yzcr_close_window = yes
        }
        else = {
            set_variable = { yzcr_pending_work = 1 }
            yzcr_dispatch_request = yes
        }
    }
}
yzcr_request_riddle = {
    if = {
        limit = { yzcr_recruitment_allowed = yes NOT = { has_country_flag = yzcr_prompt_pending } }
        if = {
            limit = { has_country_flag = RJ_recruit_show }
            clr_country_flag = RJ_recruit_show
        }
        else_if = {
            limit = { has_country_flag = yzcr_open yzcr_rj_active = yes }
            yzcr_close_window = yes
        }
        else = {
            set_variable = { yzcr_pending_work = 2 }
            yzcr_dispatch_request = yes
        }
    }
}
