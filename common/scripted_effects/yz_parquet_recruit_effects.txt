# Character tokens and traits preserve the original PARQUET recruitment contract.
# Flags prevent the debug replay path from generating either leader twice.
yz_parquet_recruit_rino = {
    if = {
        limit = { NOT = { has_country_flag = yz_parquet_rino_recruited } }
        meta_effect = {
            text = {
                generate_character = {
                    token_base = ibaraki_rino
                    name = YZ_character_name_ibaraki_rino
                    can_be_captured = no
                    gender = female
                    nationality = JAP
                    portraits = {
                        civilian = {
                            large = "gfx/leader/ibaraki_rino.png"
                            small = "gfx/leader/ibaraki_rino_small.png"
                        }
                    }
                    allowed_civil_war = { ROOT = { is_ai = no } }
                    corps_commander = {
                        desc = ibaraki_rino_corps_desc
                        traits = {
                            yz_yuzusoft_character
                            ibaraki_rino_only_at_night
                            bmi_device
                        }
                        skill = 5
                        attack_skill = 7
                        defense_skill = 3
                        planning_skill = 5
                        logistics_skill = 3
                    }
                }
            }
        }
        set_country_flag = yz_parquet_rino_recruited
    }
}
