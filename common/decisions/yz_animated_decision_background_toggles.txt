# Player-side presentation preferences for every decision category with an
# animated description background. Absent flags preserve the existing dynamic
# default for new and old saves; each zero-cost decision flips only its owner.

add_youzi_general_category = {
    yz_toggle_recruit_category_background = {
        name = yz_toggle_animated_decision_background
        desc = yz_toggle_animated_decision_background_desc
        icon = GFX_Category_decision
        priority = 1000000
        cost = 0
        visible = {
            is_ai = no
            or = {
                has_country_flag = qiyong_youzi_1
                has_country_flag = qiyong_youzi_2
                has_country_flag = qiyong_youzi_3
            }
        }
        complete_effect = {
            if = {
                limit = { has_country_flag = yz_recruit_category_static_background }
                clr_country_flag = yz_recruit_category_static_background
            }
            else = { set_country_flag = yz_recruit_category_static_background }
        }
        ai_will_do = { factor = 0 }
    }
}
